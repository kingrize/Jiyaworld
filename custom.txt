    <script>
      // Parse Tako rankings variable
      let rankings = [];
      try {
        rankings = JSON.parse("{{rankings}}");
        if (!Array.isArray(rankings)) rankings = [];
      } catch (e) {
        rankings = [];
      }

      // Configuration
      const MAX_ITEMS = 5;

      const listEl = document.getElementById("lb-list");
      const footerEl = document.getElementById("lb-footer");

      function escapeHtml(str) {
        return String(str)
          .replace(/&/g, "&amp;")
          .replace(/</g, "&lt;")
          .replace(/>/g, "&gt;")
          .replace(/"/g, "&quot;")
          .replace(/'/g, "&#039;");
      }

      function renderEmpty() {
        listEl.innerHTML = '<div class="empty">No supporters yet</div>';
        footerEl.style.display = "none";
      }

      function renderList(items) {
        const rows = items.slice(0, MAX_ITEMS).map((r) => {
          const name = escapeHtml(r.name || "Anonymous");
          const amount = escapeHtml(r.formattedAmount || "");
          const picture = r.picture ? escapeHtml(r.picture) : "";
          const badgesRaw = (r.badges || "").split(",").map((b) => b.trim()).filter(Boolean);

          // Limit to 2 badges maximum
          const badges = badgesRaw.slice(0, 2).map((b) => 
            `<span class="badge">${escapeHtml(b)}</span>`
          ).join("");

          const avatar = picture
            ? `<img class="avatar" src="${picture}" alt="" loading="lazy" />`
            : `<div class="avatar" aria-hidden="true"></div>`;

          return `
            <div class="row">
              ${avatar}
              <div class="info">
                <p class="name">${name}</p>
                ${badges ? `<div class="badges">${badges}</div>` : ""}
              </div>
              ${amount ? `<span class="amount">${amount}</span>` : ""}
            </div>
          `;
        });

        listEl.innerHTML = rows.join("");
        footerEl.style.display = "block";
      }

      if (!rankings.length) {
        renderEmpty();
      } else {
        renderList(rankings);
      }
    </script>
